"""
AulaPay DevSecOps Security Gate.

Evalua un conjunto normalizado de hallazgos contra la politica
de seguridad del proyecto.

El mismo motor permite comparar los estados BEFORE y AFTER
sin modificar las evidencias historicas.
"""

import json
import sys
from pathlib import Path


BASE_PATH = Path("/workspace/security/gate")
POLICY_PATH = BASE_PATH / "gate-policy.json"


def load_json(path: Path) -> dict:
    """Carga un documento JSON."""
    with path.open("r", encoding="utf-8") as file:
        return json.load(file)


def resolve_findings_path() -> Path:
    """Obtiene el archivo de hallazgos indicado por argumento."""

    if len(sys.argv) > 1:
        requested_path = Path(sys.argv[1])

        if requested_path.is_absolute():
            return requested_path

        return Path("/workspace") / requested_path

    return BASE_PATH / "before-findings.json"


def is_resolved(finding: dict) -> bool:
    """
    Determina si el hallazgo tiene una decision final de remediacion
    o cierre.
    """

    status = finding.get("status", "").upper()
    decision = finding.get("decision", "").upper()

    resolved_statuses = {
        "RESOLVED",
        "FIXED",
        "CLOSED",
    }

    resolved_decisions = {
        "FIXED",
        "REMOVED",
        "ACCEPTED",
        "FALSE_POSITIVE",
    }

    return (
        status in resolved_statuses
        or decision in resolved_decisions
    )


def is_confirmed(finding: dict) -> bool:
    """
    Determina si el hallazgo fue confirmado durante el analisis.

    Soporta tanto el formato historico BEFORE como el formato AFTER.
    """

    status = finding.get("status", "").upper()
    validation = finding.get("validation", "").upper()

    return (
        status == "CONFIRMED"
        or validation == "CONFIRMED"
    )


def evaluate_finding(
    finding: dict,
    policy: dict,
) -> list[str]:
    """Evalua un hallazgo y retorna las razones de bloqueo."""

    reasons = []

    finding_id = finding.get("id", "UNKNOWN")
    severity = finding.get("severity", "").upper()
    control = finding.get("control", "").upper()
    decision = finding.get("decision", "").upper()

    confirmed = is_confirmed(finding)
    resolved = is_resolved(finding)

    if (
        severity == "CRITICAL"
        and confirmed
        and not resolved
        and policy["blocking_policy"]["critical_confirmed"]
    ):
        reasons.append(
            f"{finding_id}: CRITICAL confirmado y no resuelto."
        )

    if (
        severity == "HIGH"
        and confirmed
        and not resolved
        and policy["blocking_policy"]["high_confirmed"]
    ):
        reasons.append(
            f"{finding_id}: HIGH confirmado y no resuelto."
        )

    if (
        control == "API_SECURITY"
        and not resolved
        and policy["blocking_policy"]["api_authorization_failure"]
    ):
        reasons.append(
            f"{finding_id}: fallo de autorizacion API no resuelto."
        )

    if severity == "MEDIUM" and not resolved:
        medium_policy = policy.get("medium_policy", {})

        if medium_policy.get("requires_decision", False):
            allowed_decisions = {
                value.upper()
                for value in medium_policy.get(
                    "allowed_decisions",
                    [],
                )
            }

            if decision not in allowed_decisions:
                reasons.append(
                    f"{finding_id}: MEDIUM sin decision aceptada."
                )

    return reasons


def main() -> int:
    """Ejecuta la evaluacion del Security Gate."""

    findings_path = resolve_findings_path()

    policy = load_json(POLICY_PATH)
    assessment = load_json(findings_path)

    findings = assessment.get("findings", [])

    print("AulaPay DevSecOps - Security Gate")
    print("=" * 55)
    print(f"Input: {findings_path.name}")
    print(
        f"Assessment: "
        f"{assessment.get('assessment', 'UNSPECIFIED')}"
    )
    print()

    blocked_reasons = []

    for finding in findings:
        reasons = evaluate_finding(
            finding=finding,
            policy=policy,
        )

        if reasons:
            for reason in reasons:
                print(f"BLOCK: {reason}")
                blocked_reasons.append(reason)
        else:
            print(
                f"PASS: {finding.get('id')} "
                f"- {finding.get('status')}"
            )

    print()
    print("=" * 55)

    if blocked_reasons:
        print(
            f"Blocking conditions detected: "
            f"{len(blocked_reasons)}"
        )
        print("FINAL GATE DECISION: BLOCK")
        return 1

    print("Blocking conditions detected: 0")
    print("FINAL GATE DECISION: PASS")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
