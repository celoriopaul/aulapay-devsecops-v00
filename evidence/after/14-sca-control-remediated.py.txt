"""
Control SCA reproducible para AulaPay.

Evalua las dependencias declaradas para determinar si la version
utilizada se encuentra dentro de las versiones afectadas conocidas.

El laboratorio utiliza un advisory simulado con fines academicos.
"""

import json
from pathlib import Path


MANIFEST_PATH = Path("/workspace/security/sca/dependencies.json")


def load_manifest() -> dict:
    """Carga el manifiesto de dependencias del proyecto."""
    with MANIFEST_PATH.open(
        "r",
        encoding="utf-8",
    ) as file:
        return json.load(file)


def find_vulnerable_dependencies(
    manifest: dict,
) -> list[dict]:
    """
    Retorna dependencias cuya version actual aparece declarada
    dentro de affected_versions.
    """

    findings = []

    for dependency in manifest.get("dependencies", []):
        current_version = dependency.get("version")
        affected_versions = dependency.get(
            "affected_versions",
            [],
        )

        if current_version in affected_versions:
            findings.append(dependency)

    return findings


def print_dependency_status(
    dependency: dict,
) -> None:
    """Muestra el estado evaluado de una dependencia."""

    current_version = dependency.get("version")
    affected_versions = dependency.get(
        "affected_versions",
        [],
    )

    vulnerable = current_version in affected_versions

    print(f"Dependency: {dependency.get('name')}")
    print(f"Current version: {current_version}")
    print(f"Advisory: {dependency.get('advisory')}")
    print(f"Severity: {dependency.get('severity')}")
    print(
        "Affected versions: "
        + ", ".join(affected_versions)
    )
    print(f"Fixed version: {dependency.get('fixed_version')}")
    print(f"Usage: {dependency.get('usage')}")
    print(f"Simulated advisory: {dependency.get('simulated')}")
    print(
        "Status: "
        + ("VULNERABLE" if vulnerable else "NOT VULNERABLE")
    )
    print()


def main() -> int:
    """Ejecuta el control SCA."""

    manifest = load_manifest()

    print("AulaPay DevSecOps - SCA Dependency Analysis")
    print("=" * 55)
    print()

    for dependency in manifest.get("dependencies", []):
        print_dependency_status(dependency)

    findings = find_vulnerable_dependencies(manifest)

    print("=" * 55)

    if findings:
        print(
            f"Vulnerable dependencies detected: {len(findings)}"
        )
        print("SCA RESULT: FAIL")
        return 1

    print("Vulnerable dependencies detected: 0")
    print("SCA RESULT: PASS")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
