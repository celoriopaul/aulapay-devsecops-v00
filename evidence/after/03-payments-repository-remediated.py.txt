"""
Repositorio de acceso a datos para pagos.

La capa repository centraliza las operaciones SQL relacionadas con
pagos y mantiene separados los detalles de persistencia del resto
de la aplicación.

Esta versión utiliza consultas parametrizadas para evitar que datos
controlados por el usuario modifiquen la estructura de las sentencias
SQL ejecutadas por PostgreSQL.
"""

from asyncpg import Record

from app.core.logging import get_logger
from app.db.connection import database


logger = get_logger(__name__)


class PaymentRepository:
    """
    Gestiona las operaciones de lectura relacionadas con pagos.
    """

    async def get_by_student_code(
        self,
        student_code: str,
    ) -> list[Record]:
        """
        Obtiene los pagos asociados al código de un estudiante.

        student_code se transmite a PostgreSQL como parámetro separado
        de la sentencia SQL mediante asyncpg.
        """

        query = (
            "SELECT "
            "p.id, "
            "s.student_code AS student_id, "
            "p.external_reference, "
            "p.amount, "
            "p.currency, "
            "p.status, "
            "p.payment_date, "
            "r.storage_path AS receipt_url "
            "FROM payments p "
            "INNER JOIN students s ON s.id = p.student_id "
            "LEFT JOIN payment_receipts r ON r.payment_id = p.id "
            "WHERE s.student_code = $1 "
            "ORDER BY p.payment_date DESC"
        )

        logger.info(
            "Consultando pagos para student_code=%s",
            student_code,
        )

        async with database.pool.acquire() as connection:
            records = await connection.fetch(
                query,
                student_code,
            )

        return records


payment_repository = PaymentRepository()
